cmake_minimum_required(VERSION 3.20)
project(MotorLab VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(MOTORLAB_BUILD_GUI "Build the desktop interface" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
if(MOTORLAB_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
endif()
add_library(motorlab_core src/protocol.cpp src/session.cpp src/catalog.cpp src/motor_profiles.cpp src/cpr.cpp src/hexfile.cpp src/robocan.cpp src/firmware.cpp src/basicvm.cpp src/cansdo.cpp data/resources.qrc)
target_include_directories(motorlab_core PUBLIC src)
target_link_libraries(motorlab_core PUBLIC Qt6::Core)
target_compile_options(motorlab_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(motorlab_control src/control.cpp src/gui_backend.cpp src/continuous.cpp src/control_host.h)
target_link_libraries(motorlab_control PUBLIC motorlab_core Qt6::Network)
target_compile_options(motorlab_control PRIVATE -Wall -Wextra -Wpedantic)
add_executable(motorlabd src/cli_main.cpp src/headless_host.cpp src/firmware_job.cpp)
target_link_libraries(motorlabd PRIVATE motorlab_control)
target_compile_options(motorlabd PRIVATE -Wall -Wextra -Wpedantic)
if(MOTORLAB_BUILD_GUI)
  add_executable(motorlab src/main.cpp src/remote_session.cpp src/gui_control_host.cpp src/window.cpp src/config_browser.cpp src/config_text.cpp src/motor_profile_ui.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(motorlab PRIVATE motorlab_control Qt6::Widgets)
  target_compile_options(motorlab PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE motorlab_core Qt6::Test util)
  add_test(NAME protocol_tests COMMAND protocol_tests)
  add_executable(feature_tests tests/feature_tests.cpp)
  target_link_libraries(feature_tests PRIVATE motorlab_core Qt6::Test util)
  target_compile_definitions(feature_tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME feature_tests COMMAND feature_tests)
  add_executable(headless_tests tests/headless_tests.cpp src/headless_host.cpp src/firmware_job.cpp)
  target_link_libraries(headless_tests PRIVATE motorlab_control Qt6::Test util)
  add_test(NAME headless_tests COMMAND headless_tests)
  if(MOTORLAB_BUILD_GUI)
  add_executable(shared_backend_tests tests/shared_backend_tests.cpp src/remote_session.cpp src/window.cpp src/config_browser.cpp src/config_text.cpp src/motor_profile_ui.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(shared_backend_tests PRIVATE motorlab_core Qt6::Network Qt6::Widgets Qt6::Test util)
  add_dependencies(shared_backend_tests motorlabd)
  add_test(NAME shared_backend_tests COMMAND shared_backend_tests)
  set_tests_properties(shared_backend_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MOTORLAB_COMPILER=${CMAKE_CURRENT_SOURCE_DIR}/libexec/compiler/motorlab-compiler")
  add_executable(ui_tests tests/ui_tests.cpp src/window.cpp src/config_browser.cpp src/config_text.cpp src/motor_profile_ui.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(ui_tests PRIVATE motorlab_core Qt6::Widgets Qt6::Test util)
  target_compile_definitions(ui_tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME ui_tests COMMAND ui_tests)
  add_executable(control_tests tests/control_tests.cpp src/gui_control_host.cpp src/window.cpp src/config_browser.cpp src/config_text.cpp src/motor_profile_ui.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(control_tests PRIVATE motorlab_control Qt6::Widgets Qt6::Network Qt6::Test util)
  add_test(NAME control_tests COMMAND control_tests)
  set_tests_properties(control_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MOTORLAB_COMPILER=${CMAKE_CURRENT_SOURCE_DIR}/libexec/compiler/motorlab-compiler")
  set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MOTORLAB_COMPILER=${CMAKE_CURRENT_SOURCE_DIR}/libexec/compiler/motorlab-compiler")
  endif()
endif()
install(TARGETS motorlabd RUNTIME DESTINATION bin)
if(MOTORLAB_BUILD_GUI)
  install(TARGETS motorlab RUNTIME DESTINATION bin)
endif()
