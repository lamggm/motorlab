cmake_minimum_required(VERSION 3.20)
project(MotorLab VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(motorlab_core src/protocol.cpp src/session.cpp src/catalog.cpp src/cpr.cpp src/hexfile.cpp src/robocan.cpp src/firmware.cpp src/basicvm.cpp src/cansdo.cpp data/resources.qrc)
target_include_directories(motorlab_core PUBLIC src)
target_link_libraries(motorlab_core PUBLIC Qt6::Core)
target_compile_options(motorlab_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(motorlab src/main.cpp src/control.cpp src/continuous.cpp src/window.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
target_link_libraries(motorlab PRIVATE motorlab_core Qt6::Widgets Qt6::Network)
target_compile_options(motorlab PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE motorlab_core Qt6::Test util)
  add_test(NAME protocol_tests COMMAND protocol_tests)
  add_executable(feature_tests tests/feature_tests.cpp)
  target_link_libraries(feature_tests PRIVATE motorlab_core Qt6::Test util)
  target_compile_definitions(feature_tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME feature_tests COMMAND feature_tests)
  add_executable(ui_tests tests/ui_tests.cpp src/window.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(ui_tests PRIVATE motorlab_core Qt6::Widgets Qt6::Test util)
  target_compile_definitions(ui_tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME ui_tests COMMAND ui_tests)
  add_executable(control_tests tests/control_tests.cpp src/control.cpp src/continuous.cpp src/window.cpp src/features.cpp src/firmware_ui.cpp src/script_ui.cpp src/compiler_ui.cpp src/cpr_ui.cpp src/can_ui.cpp src/chart.cpp)
  target_link_libraries(control_tests PRIVATE motorlab_core Qt6::Widgets Qt6::Network Qt6::Test util)
  add_test(NAME control_tests COMMAND control_tests)
  set_tests_properties(control_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MOTORLAB_COMPILER=${CMAKE_CURRENT_SOURCE_DIR}/libexec/compiler/motorlab-compiler")
  set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MOTORLAB_COMPILER=${CMAKE_CURRENT_SOURCE_DIR}/libexec/compiler/motorlab-compiler")
endif()
install(TARGETS motorlab RUNTIME DESTINATION bin)
